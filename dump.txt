@dataclass(frozen=True)
class ActionResult:
    action: DecisionAction
    success: bool
    message: str


class ActionExecutor:
    def __init__(
        self,
        alert_handler: Callable[[Decision], bool] | None = None,
        protect_handler: Callable[[Decision], bool] | None = None,
        isolate_handler: Callable[[Decision], bool] | None = None,
        recover_handler: Callable[[Decision], bool] | None = None,
        audit_log: DecisionAuditLog | None = None,
    ):
        self.alert_handler = alert_handler
        self.protect_handler = protect_handler
        self.isolate_handler = isolate_handler
        self.recover_handler = recover_handler
        self.audit_log = audit_log

    def execute(self, decision: Decision) -> ActionResult:
        if decision.action == DecisionAction.ALLOW:
            result = ActionResult(
                action=decision.action,
                success=True,
                message="Event allowed.",
            )
        else:
            handlers = {
                DecisionAction.ALERT: self.alert_handler,
                DecisionAction.PROTECT: self.protect_handler,
                DecisionAction.ISOLATE: self.isolate_handler,
                DecisionAction.RECOVER: self.recover_handler,
            }

            handler = handlers.get(decision.action)

            if handler is None:
                result = ActionResult(
                    action=decision.action,
                    success=False,
                    message=(
                        f"No handler configured for "
                        f"{decision.action.value}."
                    ),
                )
            else:
                try:
                    success = bool(handler(decision))

                    result = ActionResult(
                        action=decision.action,
                        success=success,
                        message=(
                            f"{decision.action.value} action completed."
                            if success
                            else f"{decision.action.value} action failed."
                        ),
                    )

                except Exception as error:
                    result = ActionResult(
                        action=decision.action,
                        success=False,
                        message=(
                            f"{decision.action.value} action raised an error: "
                            f"{error}"
                        ),
                    )

        if self.audit_log is not None:
            self.audit_log.record(
                decision,
                success=result.success,
                message=result.message,
            )

        return resultfrom dataclasses import asdict, dataclass

@dataclass(frozen=True)
class DecisionAudit:
    event_id: str
    host_id: str
    action: DecisionAction
    risk_level: str
    risk_score: float
    reason: str
    success: bool
    message: str
    recorded_at: str

    def to_dict(self) -> dict:
        return asdict(self)


class DecisionAuditLog:
    def __init__(self):
        self._records: list[DecisionAudit] = []

    def record(
        self,
        decision: Decision,
        success: bool,
        message: str,
    ) -> DecisionAudit:
        record = DecisionAudit(
            event_id=decision.event_id,
            host_id=decision.host_id,
            action=decision.action,
            risk_level=decision.risk_level.value,
            risk_score=decision.risk_score,
            reason=decision.reason,
            success=success,
            message=message,
            recorded_at=datetime.now(timezone.utc).isoformat(),
        )

        self._records.append(record)

        return record

    def all(self) -> list[DecisionAudit]:
        return list(self._records)

    def for_event(self, event_id: str) -> list[DecisionAudit]:
        return [
            record
            for record in self._records
            if record.event_id == event_id
        ]from backend.app.decision.actions import DecisionAction

def test_isolation_action_calls_isolation_handler():
    executed = []

    def isolate(decision):
        executed.append(decision.host_id)
        return True

    event = SecurityEvent(
        event_id="event-008",
        host_id="host-02",
        threat_type="ransomware",
        risk_level=RiskLevel.CRITICAL,
        risk_score=97.0,
    )

    decision = DecisionEngine().decide(event)

    result = ActionExecutor(
        isolate_handler=isolate,
    ).execute(decision)

    assert result.action == DecisionAction.ISOLATE
    assert result.success is True
    assert executed == ["host-02"]

def test_action_execution_creates_audit_record():
    audit_log = DecisionAuditLog()

    def isolate(decision):
        return True

    event = SecurityEvent(
        event_id="event-013",
        host_id="host-07",
        threat_type="ransomware",
        risk_level=RiskLevel.CRITICAL,
        risk_score=98.0,
    )

    decision = DecisionEngine().decide(event)

    result = ActionExecutor(
        isolate_handler=isolate,
        audit_log=audit_log,
    ).execute(decision)

    assert result.success is True

    records = audit_log.for_event("event-013")

    assert len(records) == 1
    assert records[0].action == DecisionAction.ISOLATE
    assert records[0].success is True

def create_decision():
    event = SecurityEvent(
        event_id="event-100",
        host_id="host-01",
        threat_type="ransomware_behavior",
        risk_level=RiskLevel.CRITICAL,
        risk_score=96.0,
    )

    return DecisionEngine().decide(event)


def test_audit_records_decision():
    decision = create_decision()
    audit_log = DecisionAuditLog()

    record = audit_log.record(
        decision,
        success=True,
        message="Host isolated successfully.",
    )

    assert record.event_id == "event-100"
    assert record.host_id == "host-01"
    assert record.action == DecisionAction.ISOLATE
    assert record.risk_level == "CRITICAL"
    assert record.risk_score == 96.0
    assert record.success is True
    assert record.message == "Host isolated successfully."
    assert record.recorded_at


def test_audit_returns_all_records():
    decision = create_decision()
    audit_log = DecisionAuditLog()

    audit_log.record(
        decision,
        success=True,
        message="Host isolated.",
    )

    records = audit_log.all()

    assert len(records) == 1
    assert records[0].event_id == "event-100"


def test_audit_can_filter_by_event():
    decision = create_decision()
    audit_log = DecisionAuditLog()

    audit_log.record(
        decision,
        success=True,
        message="Host isolated.",
    )

    records = audit_log.for_event("event-100")

    assert len(records) == 1
    assert records[0].action == DecisionAction.ISOLATE


def test_audit_unknown_event_returns_empty_list():
    audit_log = DecisionAuditLog()

    records = audit_log.for_event("does-not-exist")

    assert records == []


def test_audit_record_can_be_serialized():
    decision = create_decision()
    audit_log = DecisionAuditLog()

    record = audit_log.record(
        decision,
        success=False,
        message="Isolation failed.",
    )

    data = record.to_dict()

    assert data["event_id"] == "event-100"
    assert data["action"] == DecisionAction.ISOLATE
    assert data["success"] is False
